### 3. Экспорт в CSV/Excel
```rust
use csv::Writer;

pub fn export_to_csv(results: &ScanResults, path: &PathBuf) -> Result<()> {
    let mut wtr = Writer::from_path(path)?;
    
    wtr.write_record(&["Path", "Size", "Extension", "Modified"])?;
    
    for file in &results.files {
        wtr.write_record(&[
            file.path.to_str().unwrap(),
            &file.size.to_string(),
            &file.extension,
            &file.modified.to_string(),
        ])?;
    }
    
    wtr.flush()?;
    Ok(())
}
```

**Зависимость**: `csv = "1.3"`
